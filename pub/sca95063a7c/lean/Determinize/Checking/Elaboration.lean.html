7private def actionAffinities : DistributionAction → List Affinity
8 | .sample affinity => [affinity]
9 | .mean => []

11def sampleAffinities : Core → List Affinity
12 | .bvar _ => []
13 | .reject => []
14 | .unit => []
15 | .bool _ => []
16 | .real _ => []
17 | .lam body => (sampleAffinities body)
18 | .fix body => (sampleAffinities body)
19 | .app fn arg => (sampleAffinities fn) ++ (sampleAffinities arg)
20 | .pair left right => (sampleAffinities left) ++ (sampleAffinities right)
21 | .fst pairValue => (sampleAffinities pairValue)
22 | .snd pairValue => (sampleAffinities pairValue)
23 | .inl value => (sampleAffinities value)
24 | .inr value => (sampleAffinities value)
25 | .matchSum scrutinee left right => (sampleAffinities scrutinee) ++ (sampleAffinities left) ++ (sampleAffinities right)
26 | .nil => []
27 | .cons head tail => (sampleAffinities head) ++ (sampleAffinities tail)
28 | .matchList scrutinee nilCase consCase => (sampleAffinities scrutinee) ++ (sampleAffinities nilCase) ++ (sampleAffinities consCase)
29 | .ite condition thenBranch elseBranch => (sampleAffinities condition) ++ (sampleAffinities thenBranch) ++ (sampleAffinities elseBranch)
30 | .letE value body => (sampleAffinities value) ++ (sampleAffinities body)
31 | .neg body => (sampleAffinities body)
32 | .add left right => (sampleAffinities left) ++ (sampleAffinities right)
33 | .mul left right => (sampleAffinities left) ++ (sampleAffinities right)
34 | .div left right => (sampleAffinities left) ++ (sampleAffinities right)
35 | .lt left right => (sampleAffinities left) ++ (sampleAffinities right)
36 | .uniform action lower upper => actionAffinities action ++ ((sampleAffinities lower) ++ (sampleAffinities upper))
37 | .gaussian action mean variance => actionAffinities action ++ ((sampleAffinities mean) ++ (sampleAffinities variance))
38 | .poisson action rate => actionAffinities action ++ ((sampleAffinities rate))
39 | .discrete action p => actionAffinities action ++ sampleAffinities p
40 | .bernoulli action probability => actionAffinities action ++ ((sampleAffinities probability))
41 | .exponential action rate => actionAffinities action ++ ((sampleAffinities rate))
42 | .beta action alpha betaArg => actionAffinities action ++ ((sampleAffinities alpha) ++ (sampleAffinities betaArg))
43 | .gamma action shape rate => actionAffinities action ++ ((sampleAffinities shape) ++ (sampleAffinities rate))

46def Input.matches : Input → Core → Bool
47 | .bvar index, .bvar index' => decide (index = index')
48 | .reject, .reject => true
49 | .unit, .unit => true
50 | .bool value, .bool value' => decide (value = value')
51 | .real value, .real value' => decide (value = value')
52 | .nil, .nil => true
53 | .lam body, .lam body' =>
54 body.matches body'
55 | .fix body, .fix body' =>
56 body.matches body'
57 | .app fn arg, .app fn' arg' =>
58 fn.matches fn' && arg.matches arg'
59 | .pair left right, .pair left' right' =>
60 left.matches left' && right.matches right'
61 | .fst body, .fst body' =>
62 body.matches body'
63 | .snd body, .snd body' =>
64 body.matches body'
65 | .inl body, .inl body' =>
66 body.matches body'
67 | .inr body, .inr body' =>
68 body.matches body'
69 | .matchSum scrutinee left right, .matchSum scrutinee' left' right' =>
70 scrutinee.matches scrutinee' && left.matches left' && right.matches right'
71 | .cons head tail, .cons head' tail' =>
72 head.matches head' && tail.matches tail'
73 | .matchList scrutinee nilCase consCase, .matchList scrutinee' nilCase' consCase' =>
74 scrutinee.matches scrutinee' && nilCase.matches nilCase' && consCase.matches consCase'
75 | .ite condition thenBranch elseBranch, .ite condition' thenBranch' elseBranch' =>
76 condition.matches condition' && thenBranch.matches thenBranch' && elseBranch.matches elseBranch'
77 | .letE value body, .letE value' body' =>
78 value.matches value' && body.matches body'
79 | .neg body, .neg body' =>
80 body.matches body'
81 | .add left right, .add left' right' =>
82 left.matches left' && right.matches right'
83 | .mul left right, .mul left' right' =>
84 left.matches left' && right.matches right'
85 | .div left right, .div left' right' =>
86 left.matches left' && right.matches right'
87 | .lt left right, .lt left' right' =>
88 left.matches left' && right.matches right'
89 | .uniform requested lower upper, .uniform (.sample actual) lower' upper' =>
90 (requested.isNone || requested == some actual) && lower.matches lower' && upper.matches upper'
91 | .gaussian requested mean variance, .gaussian (.sample actual) mean' variance' =>
92 (requested.isNone || requested == some actual) && mean.matches mean' && variance.matches variance'
93 | .poisson requested rate, .poisson (.sample actual) rate' =>
94 (requested.isNone || requested == some actual) && rate.matches rate'
95 | .bernoulli requested probability, .bernoulli (.sample actual) probability' =>
96 (requested.isNone || requested == some actual) && probability.matches probability'
97 | .exponential requested rate, .exponential (.sample actual) rate' =>
98 (requested.isNone || requested == some actual) && rate.matches rate'
99 | .beta requested alpha betaArg, .beta (.sample actual) alpha' betaArg' =>
100 (requested.isNone || requested == some actual) && alpha.matches alpha' && betaArg.matches betaArg'
101 | .gamma requested shape rate, .gamma (.sample actual) shape' rate' =>
102 (requested.isNone || requested == some actual) && shape.matches shape' && rate.matches rate'
103 | .discrete requested distribution, .discrete (.sample actual) distribution' =>
104 (requested.isNone || requested == some actual) && distribution.matches distribution'
105 | _, _ => false

107structure Certified (input : Input) where
108 source : Core
109 ty : Ty
110 typed : Typed [] (interpret source) ty
111 aligned : input.matches source = true
112
113def certify (input : Input) (candidate : Core) (c : Certificate) :
114 Option (Certified input) := do
115 if hAlign : input.matches candidate = true then
116 let h ← check [] candidate c.ty c
117 return ⟨candidate, c.ty, h.down, hAlign⟩
118 else none
