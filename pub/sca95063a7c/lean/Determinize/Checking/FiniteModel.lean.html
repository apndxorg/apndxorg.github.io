7/-- Kernel-replayable evidence of graph validity. `replay_matches` transports
8this evidence to the paper semantics. -/
9def checkModelReplay (source : Core) (subject : Subject) (candidate : Candidate) :
10 Option (PLift (candidate.ReplayValid source subject)) :=
11 if valid : candidate.ReplayValid source subject then some ⟨valid⟩ else none
12
13structure CheckedModel (source : Core) (subject : Subject) where
14 model : Model
15 correct : model.Matches (subject.program source)
16
17def checkModel (source : Core) (subject : Subject) (candidate : Candidate) :
18 Option (CheckedModel source subject) := do
19 let valid ← checkModelReplay source subject candidate
20 return ⟨candidate.toModel valid.down, Proof.FiniteModel.replay_matches candidate valid.down⟩

22end Determinize.Checking
