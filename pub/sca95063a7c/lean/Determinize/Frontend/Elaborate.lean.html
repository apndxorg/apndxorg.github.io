1import Determinize.Frontend.Syntax
2import Determinize.Frontend.Shift
3import Determinize.Checking.FiniteDistribution

5namespace Determinize.Frontend
6open Spec.Paper Checking
7
8private def index (env : List String) (x : String) : Except String Nat :=
9 match env with
10 | [] => .error s!"unbound variable '{x}'"
11 | y :: ys => if x == y then .ok 0 else (· + 1) <$> index ys x
12
13private def lower (env : List String) : Surface → Except String Input
14 | .var x => do pure (.bvar (← index env x))
15 | .number q => pure (.real q)
16 | .lam x b => return .lam (← lower (x :: env) b)
17 | .fix f x b => return .fix (← lower (x :: f :: env) b)
18 | .letE x a b => return .letE (← lower env a) (← lower (x :: env) b)
19 | .matchList x xs e n c =>
20 return .matchList (← lower env e) (← lower env n) (← lower (x :: xs :: env) c)
21 | .matchSum x y e l r =>
22 return .matchSum (← lower env e) (← lower (x :: env) l) (← lower (y :: env) r)
23 | .unit => return .unit
24 | .nil => return .nil
25 | .bool b => return (.bool b)
26 | .ite c a b => return .ite (← lower env c) (← lower env a) (← lower env b)
27 | .discrete affinity args => do
28 let weights ← args.mapM fun a => match a with
29 | .number q => pure q
30 | _ => throw "discrete expects nonnegative literal weights"
31 let distribution ← finiteDistribution weights
32 return .discrete affinity (distribution.probabilities.dropLast.foldr
33 (fun p tail => .cons (.real p) tail) .nil)
34 | .discreteRemainder affinity probabilities =>
35 return .discrete affinity (← lower env probabilities)
36 | .neg a => return .neg (← lower env a)
37 | .fst a => return .fst (← lower env a)
38 | .snd a => return .snd (← lower env a)
39 | .inl a => return .inl (← lower env a)
40 | .inr a => return .inr (← lower env a)
41 | .poisson affinity a => return .poisson affinity (← lower env a)
42 | .exponential affinity a => return .exponential affinity (← lower env a)
43 | .bernoulli affinity a => return .bernoulli affinity (← lower env a)
44 | .observe condition => do
45 let a ← lower env condition
46 return .ite a .unit .reject
47 | .flip affinity probability => do
48 if affinity == some .E then throw "flip produces a Boolean and requires [G]"
49 let draw : Input := .bernoulli (some .G) (← lower env probability)
50 return .lt (.real 0) draw
51 | .app a b => return .app (← lower env a) (← lower env b)
52 | .pair a b => return .pair (← lower env a) (← lower env b)
53 | .cons a b => return .cons (← lower env a) (← lower env b)
54 | .add a b => return .add (← lower env a) (← lower env b)
55 | .div a b => return .div (← lower env a) (← lower env b)
56 | .lt a b => return .lt (← lower env a) (← lower env b)
57 | .sub a b => return .add (← lower env a) (.neg (← lower env b))
58 | .mul a b => do
59 let a ← lower env a; let b ← lower env b
60 match a, b with
61 | .real _, _ => return .mul a b
62 | _, .real _ => return .mul b a
63 | _, _ => return .mul a b
64 | .le a b => do
65 let a ← lower env a; let b ← lower env b
66 return .letE a (.letE
67 (b.shift 1 0) (.ite (.lt (.bvar 0) (.bvar 1)) (.bool false) (.bool true)))
68 | .uniform affinity a b => return .uniform affinity (← lower env a) (← lower env b)
69 | .gaussian affinity a b => return .gaussian affinity (← lower env a) (← lower env b)
70 | .beta affinity a b => return .beta affinity (← lower env a) (← lower env b)
71 | .gamma affinity a b => return .gamma affinity (← lower env a) (← lower env b)
72
73def elaborate (e : Surface) : Except String Input := lower [] e

75end Determinize.Frontend
