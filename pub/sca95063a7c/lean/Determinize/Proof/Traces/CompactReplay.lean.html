4/-!
5# The compact replay as a measurable kernel
6
7`Proof.Traces.outputGivenTraceAt` replays a program along a compact trace of general-affinity draws.
8This file packages it as an s-finite kernel in the trace and the expression
9(`compactReplayKernel`), proves the one-step unfolding lemmas the lockstep arguments use, bounds
10the total mass of `Proof.Traces.outputGivenTrace` by one, and builds the Markov kernel that agrees
11with it wherever it has mass one. Lemmas about `Proof.Traces.outputGivenTraceAt` are named
12`ogtAt_*` throughout this file.
13-/

152/-- The active general-affinity site of the expression fits the head of the tape. -/
153def matchesHead (pair : DrawTrace × Expr) : Prop :=
154 pair.1 ≠ [] ∧ generationOp pair.2.skeleton = some (pair.1.getD 0 (.uniform, 0)).1

156theorem nilRegion_measurable : MeasurableSet {pair : DrawTrace × Expr | pair.1 = []} := by
157 have eq : {pair : DrawTrace × Expr | pair.1 = []} =
158 (fun pair : DrawTrace × Expr => pair.1.length) ⁻¹' {0} := by

164theorem matchesHead_measurable : MeasurableSet {pair : DrawTrace × Expr | matchesHead pair} := by
165 have eq : {pair : DrawTrace × Expr | matchesHead pair} =
166 {pair : DrawTrace × Expr | pair.1 = []}ᶜ ∩

177/-- One replay step: a deterministic step or an expectation-affinity draw leaves the tape alone;
178a general-affinity draw consumes the head of the tape when its primitive fits, and stops
179otherwise. -/
180def compactReplayStep : SFiniteKernel (DrawTrace × Expr) (DrawTrace × Expr) := by
181 let step := MeasurableActionFamily.stepKernel primitiveLaws
182 let sampled := SFiniteKernel.mapWithInput
183 (SFiniteKernel.pullback ⟨step.kernel, step.kernel_sfinite⟩ Prod.snd measurable_snd)
184 (fun pair : (DrawTrace × Expr) × Expr => (pair.1.1, pair.2))
185 ((measurable_fst.comp measurable_fst).prodMk measurable_snd)
186 let forced := SFiniteKernel.deterministic
187 (fun pair : DrawTrace × Expr =>
188 (List.tail pair.1, sampleContinuation pair.2 (pair.1.getD 0 (.uniform, 0)).2))
189 ((draw_tail_measurable.comp measurable_fst).prodMk
190 (sampleContinuation_measurable.comp (measurable_snd.prodMk
191 (measurable_snd.comp ((draw_event_measurable 0).comp measurable_fst)))))
192 let region : Set (DrawTrace × Expr) := {pair | generationOp pair.2.skeleton = none}
193 have measurableRegion : MeasurableSet region :=
194 (measurable_skeleton.comp measurable_snd)
195 (show MeasurableSet {s | generationOp s = none} from trivial)
196 exact SFiniteKernel.piecewise measurableRegion sampled
197 (SFiniteKernel.piecewise matchesHead_measurable forced SFiniteKernel.zero)

199theorem compactReplayStep_apply (tape : DrawTrace) (expression : Expr) :
200 compactReplayStep.kernel (tape, expression) =
201 if generationOp expression.skeleton = none then
202 (Determinize.Spec.Paper.stepMeasure expression).map (fun next => (tape, next))
203 else if matchesHead (tape, expression) then
204 Measure.dirac (tape.tail, sampleContinuation expression (tape.getD 0 (.uniform, 0)).2)
205 else 0 := by
206 unfold compactReplayStep SFiniteKernel.piecewise
207 simp only [Kernel.piecewise, Kernel.coe_mk, Set.mem_ofPred_eq]
208 by_cases none : generationOp expression.skeleton = none
209 · rw [if_pos none, if_pos none,
210 SFiniteKernel.mapWithInput_apply,
211 MeasurableActionFamily.pullback_apply, StepKernel.kernel_eq_stepMeasure]
212 · rw [if_neg none, if_neg none]
213 by_cases fits : matchesHead (tape, expression)
214 · rw [if_pos fits, if_pos fits]
215 rfl
216 · rw [if_neg fits, if_neg fits]
217 rfl

219/-- The compact replay at a fixed depth, jointly measurable in the tape and the expression. -/
220def compactReplayKernel : (depth : Nat) → SFiniteKernel (DrawTrace × Expr) ℝ
221 | 0 => SFiniteKernel.piecewise nilRegion_measurable
222 (SFiniteKernel.pullback
223 (MeasurableActionFamily.exactOutputKernelPack
224 (MeasurableActionFamily.stepKernel primitiveLaws) 0) Prod.snd measurable_snd)
225 SFiniteKernel.zero
226 | depth + 1 => by
227 let previous := compactReplayKernel depth
228 let step := compactReplayStep
229 letI := previous.sfinite
230 letI := step.sfinite
231 have measurableValues : MeasurableSet {pair : DrawTrace × Expr | pair.2.isValue = true} :=
232 MeasurableActionFamily.valueSet_measurable.preimage measurable_snd
233 exact SFiniteKernel.piecewise measurableValues SFiniteKernel.zero
234 ⟨previous.kernel ∘ₖ step.kernel, inferInstance⟩

236theorem compactReplayKernel_apply (depth : Nat) (tape : DrawTrace) (expression : Expr) :
237 (compactReplayKernel depth).kernel (tape, expression) =
238 outputGivenTraceAt depth expression tape := by
239 induction depth generalizing expression tape with
240 | zero =>
241 unfold compactReplayKernel SFiniteKernel.piecewise
242 simp only [Kernel.piecewise, Kernel.coe_mk, Set.mem_ofPred_eq]
243 by_cases nil : tape = []
244 · rw [if_pos nil, MeasurableActionFamily.pullback_apply]
245 subst nil
246 exact (MeasurableActionFamily.exactOutputKernel_apply _ _ _).trans
247 (by cases expression <;> rfl)
248 · rw [if_neg nil]
249 obtain ⟨head, rest, rfl⟩ := List.exists_cons_of_ne_nil nil
250 exact (ogtAt_zero_cons _ _ _).symm
251 | succ depth ih =>
252 unfold compactReplayKernel SFiniteKernel.piecewise
253 dsimp only
254 simp only [Kernel.piecewise, Kernel.coe_mk, Set.mem_ofPred_eq]
255 by_cases value : expression.isValue = true
256 · rw [if_pos value, ogtAt_succ_value depth value]
257 rfl
258 · rw [if_neg value, Kernel.comp_apply, compactReplayStep_apply]
259 let previous := SFiniteKernel.pullback (compactReplayKernel depth)
260 (fun next : Expr => (tape, next)) (measurable_const.prodMk measurable_id)
261 have previousEq : ∀ next, previous.kernel next = outputGivenTraceAt depth next tape := by
262 intro next
263 rw [MeasurableActionFamily.pullback_apply, ih]
264 have previousFun : (fun next : Expr => (compactReplayKernel depth).kernel (tape, next)) =
265 previous.kernel := by
266 funext next
267 rw [MeasurableActionFamily.pullback_apply]
268 by_cases none : generationOp expression.skeleton = none
269 · rw [if_pos none, bind_map _ (fun next : Expr => (tape, next))
270 (measurable_const.prodMk measurable_id) (compactReplayKernel depth).kernel, previousFun]
271 unfold Determinize.Spec.Paper.stepMeasure
272 cases reduction : reduce expression with
273 | next next =>
274 rw [Action.measure, Measure.dirac_bind previous.kernel.measurable, previousEq,
275 ogtAt_succ_next depth value reduction]
276 | sample site fiber continuation =>
277 rw [Action.measure, ogtAt_succ_sampleE depth value reduction
278 (by rw [← reduce_site reduction, none]),
279 bind_map _ _ (continuation_measurable reduction)]
280 simp_rw [previousEq]
281 | stuck =>
282 rw [Action.measure, Measure.bind_zero_left, ogtAt_succ_stuck depth value reduction]
283 · rw [if_neg none]
284 obtain ⟨op, active⟩ := Option.ne_none_iff_exists'.mp none
285 rcases generationOp_some_reduce active with ⟨fiber, continuation, reduction⟩ | reduction
286 · by_cases fits : matchesHead (tape, expression)
287 · rw [if_pos fits, Measure.dirac_bind (compactReplayKernel depth).kernel.measurable,
288 ih]
289 obtain ⟨⟨op', v⟩, rest, rfl⟩ := List.exists_cons_of_ne_nil fits.1
290 have opEq : op' = op := by
291 have h := fits.2
292 simp only [active, List.getD_cons_zero, Option.some.injEq] at h
293 exact h.symm
294 subst opEq
295 simp only [List.tail_cons, List.getD_cons_zero]
296 rw [ogtAt_succ_sampleG depth value reduction, sampleContinuation, reduction]
297 · rw [if_neg fits, Measure.bind_zero_left]
298 cases tape with
299 | nil => exact (ogtAt_succ_sampleG_nil depth value reduction).symm
300 | cons head rest =>
301 obtain ⟨op', v⟩ := head
302 have ne : op ≠ op' := fun h => fits ⟨List.cons_ne_nil _ _, by simp [active, h]⟩
303 exact (ogtAt_succ_sampleG_mismatch depth value reduction ne v rest).symm
304 · rw [ogtAt_succ_stuck depth value reduction]
305 split_ifs with fits
306 · rw [Measure.dirac_bind (compactReplayKernel depth).kernel.measurable, ih,
307 sampleContinuation, reduction, ogtAt_unit]
308 · exact Measure.bind_zero_left _

353theorem ogtAt_partial_mass_le_one (n : Nat) (expression : Expr) (tape : DrawTrace) :
354 ∑ depth ∈ Finset.range n, outputGivenTraceAt depth expression tape Set.univ ≤ 1 := by
355 induction n generalizing expression tape with
356 | zero => simp
357 | succ n ih =>
358 rw [Finset.sum_range_succ']
359 by_cases value : expression.isValue = true
360 · simp only [ogtAt_succ_value _ value, Measure.coe_zero, Pi.zero_apply,
361 Finset.sum_const_zero, zero_add]
362 cases expression <;> cases tape <;> simp [outputGivenTraceAt]
363 · have zero : outputGivenTraceAt 0 expression tape Set.univ = 0 := by
364 rw [ogtAt_zero_of_notReal (fun v h => value (by subst h; rfl)) tape]
365 rfl
366 rw [zero, add_zero]
367 cases reduction : reduce expression with
368 | next next =>
369 simp only [ogtAt_succ_next _ value reduction]
370 exact ih next tape
371 | stuck => simp [ogtAt_succ_stuck _ value reduction]
372 | sample site fiber continuation =>
373 rcases site with ⟨kind, op⟩
374 by_cases generation : siteOp (kind, op) = none
375 · simp only [ogtAt_succ_sampleE _ value reduction generation]
376 simp_rw [Measure.bind_apply MeasurableSet.univ
377 (ogtAt_continuation_measurable _ reduction tape).aemeasurable]
378 have coeMeasurable (depth : Nat) : Measurable
379 (fun v : ℝ => outputGivenTraceAt depth (continuation v) tape Set.univ) :=
380 Measurable.comp (MeasureTheory.Measure.measurable_coe MeasurableSet.univ)
381 (ogtAt_continuation_measurable depth reduction tape)
382 rw [← lintegral_finsetSum _ (fun depth _ => coeMeasurable depth)]
383 calc ∫⁻ v, ∑ depth ∈ Finset.range n,
384 outputGivenTraceAt depth (continuation v) tape Set.univ ∂fiber
385 ≤ ∫⁻ _, 1 ∂fiber := lintegral_mono fun v => ih (continuation v) tape
386 _ = fiber Set.univ := by simp
387 _ ≤ 1 := reduce_sample_mass_le_one reduction
388 · have general : kind = .sample .G := by
389 cases kind with
390 | sample affinity => cases affinity <;> simp_all [siteOp]
391 | mean => simp_all [siteOp]
392 subst kind
393 cases tape with
394 | nil => simp [ogtAt_succ_sampleG_nil _ value reduction]
395 | cons head rest =>

451/-- The traces on which a kernel is the Dirac mass at a measurable function form a
452measurable set. -/
453theorem measurableSet_kernel_eq_dirac {α : Type*} [MeasurableSpace α] (κ : Kernel α ℝ)
454 [IsSFiniteKernel κ] {g : α → ℝ} (hg : Measurable g) :
455 MeasurableSet {a | κ a = Measure.dirac (g a)} := by
456 have offDiagonal : MeasurableSet {p : α × ℝ | p.2 ≠ g p.1} :=
457 (measurableSet_eq_fun measurable_snd (hg.comp measurable_fst)).compl
458 have preimage (a : α) : Prod.mk a ⁻¹' {p : α × ℝ | p.2 ≠ g p.1} = {g a}ᶜ := by
459 ext v
460 simp
461 have eq : {a | κ a = Measure.dirac (g a)} = {a | κ a Set.univ = 1} ∩
462 {a | κ a (Prod.mk a ⁻¹' {p : α × ℝ | p.2 ≠ g p.1}) = 0} := by
463 ext a
464 simp only [Set.mem_ofPred_eq, Set.mem_inter_iff, preimage]
465 constructor
466 · intro h
467 rw [h]
468 exact ⟨by simp, by simp⟩
469 · rintro ⟨mass, null⟩
470 ext s hs
471 have diffNull : κ a (s \ {g a}) = 0 := measure_mono_null (fun v hv => hv.2) null
472 rw [← measure_inter_add_sdiff s (measurableSet_singleton (g a)), diffNull, add_zero,
473 Measure.dirac_apply' _ hs]
474 have singleton : κ a {g a} = 1 := by
475 have h := measure_add_measure_compl (μ := κ a) (measurableSet_singleton (g a))
476 rwa [mass, null, add_zero] at h
477 by_cases mem : g a ∈ s
478 · rw [Set.inter_eq_right.mpr (Set.singleton_subset_iff.mpr mem), singleton,
479 Set.indicator_of_mem mem]
480 rfl
481 · rw [Set.inter_singleton_eq_empty.mpr mem, measure_empty, Set.indicator_of_notMem mem]
482 rw [eq]
483 exact (measurableSet_eq_fun (κ.measurable_coe MeasurableSet.univ) measurable_const).inter
484 (measurableSet_eq_fun (Kernel.measurable_kernel_prodMk_left offDiagonal) measurable_const)
