4/-!
5# Compact traces
6
7The compact trace of a detailed step trace retains its general-affinity draws (`retain`), and the
8measurable structure of compact traces is generated by their length and coordinates.
9-/
10
11namespace Determinize.Proof.StepTraces
12open MeasureTheory ProbabilityTheory Determinize.Spec.Paper Determinize.Proof.StepTraces
13open Determinize.Proof.Paper
14open scoped ProbabilityTheory
15noncomputable section
16
17abbrev DrawTrace := Determinize.Spec.Traces.Trace
18
19theorem draw_length_measurable : Measurable (List.length : DrawTrace → Nat) :=
20 measurable_fst.comp (comap_measurable _)
21
22theorem draw_event_measurable (i : Nat) :
23 Measurable (fun trace : DrawTrace => trace.getD i (.uniform, 0)) := by
24 have h : Measurable (fun trace : DrawTrace =>
25 (trace.length, fun j : Nat => trace.getD j (.uniform, 0))) := comap_measurable _
26 exact (measurable_pi_apply i).comp (measurable_snd.comp h)
27
28theorem draw_cons_measurable :
29 Measurable (fun p : (Op × ℝ) × DrawTrace => p.1 :: p.2) := by
30 apply measurable_comap_iff.mpr
31 refine Measurable.prodMk ((draw_length_measurable.comp measurable_snd).add_const 1) ?_
32 apply measurable_pi_lambda
33 intro i
34 cases i with
35 | zero => exact measurable_fst
36 | succ i => exact (draw_event_measurable i).comp measurable_snd
37
38theorem draw_tail_measurable : Measurable (List.tail : DrawTrace → DrawTrace) := by
39 apply measurable_comap_iff.mpr
40 refine Measurable.prodMk ?_ ?_
41 · simpa using draw_length_measurable.sub_const 1
42 · apply measurable_pi_lambda
43 intro i
44 simpa using draw_event_measurable (i+1)
45
46/-- Put an event's draw, if it has one, in front of a compact trace. -/
47def emitDraw (event : Event) (trace : DrawTrace) : DrawTrace :=
48 match event with | none => trace | some draw => draw :: trace
49
50theorem event_elim_measurable {α β : Type*} [MeasurableSpace α] [MeasurableSpace β]
51 (empty : α → β) (draw : (Op × ℝ) × α → β)
52 (he : Measurable empty) (hd : Measurable draw) :
53 Measurable (fun p : Event × α => p.1.elim (empty p.2) (fun d => draw (d,p.2))) := by
54 let f : (Sum (Op × ℝ) PUnit) × α → β := fun p =>
55 match p.1 with | .inl d => draw (d,p.2) | .inr _ => empty p.2
56 have hf : Measurable f := by
57 convert (hd.sumElim (he.comp measurable_snd)).comp
58 (MeasurableEquiv.sumProdDistrib (Op × ℝ) PUnit α).measurable using 1
59 funext ⟨e,a⟩
60 cases e <;> rfl
61 have h := hf.comp ((comap_measurable (Equiv.optionEquivSumPUnit.{0} (Op × ℝ))).prodMap (measurable_id (α := α)))
62 convert h using 1
63 funext ⟨e,a⟩
64 cases e <;> rfl
65
66theorem emitDraw_measurable : Measurable (fun p : Event × DrawTrace => emitDraw p.1 p.2) := by
67 convert event_elim_measurable id (fun p => p.1 :: p.2) measurable_id draw_cons_measurable using 1
68 funext ⟨e,a⟩
69 cases e <;> rfl
70
71/-- The compact trace of the first `n` events of a detailed trace. -/
72def retainWithin : Nat → Trace → DrawTrace
73 | 0, _ => []
74 | n+1, trace => emitDraw (trace.headD none) (retainWithin n trace.tail)
75
76theorem retainWithin_measurable (n : Nat) : Measurable (retainWithin n) := by
77 induction n with
78 | zero => exact measurable_const
79 | succ n ih =>
80 exact emitDraw_measurable.comp
81 (trace_head_measurable.prodMk (ih.comp trace_tail_measurable))
82
83/-- The compact trace of a detailed trace: exactly its G-affinity draws, in order. -/
84def retain (trace : Trace) : DrawTrace := trace.filterMap id
85
86theorem retain_eq (trace : Trace) : retain trace = retainWithin trace.length trace := by
87 induction trace with
88 | nil => rfl
89 | cons head tail ih => cases head <;> simp_all [retain, retainWithin, emitDraw]
90
91theorem retain_measurable : Measurable retain := by
92 rw [funext retain_eq]
93 have h : Measurable (fun p : Nat × Trace => retainWithin p.1 p.2) :=
94 measurable_from_prod_countable_right retainWithin_measurable
95 exact h.comp (trace_length_measurable.prodMk measurable_id)
96
97end
