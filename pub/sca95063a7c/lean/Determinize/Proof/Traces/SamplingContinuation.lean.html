3/-!
4# Sampling continuations
5
6`sampleContinuation` feeds a recorded value to the sampling step of an expression; the compact
7replay (`Proof/Traces/CompactReplay.lean`) and the symbolic trace generation use it to read a
8G-affinity draw from a trace, and `sampleContinuation_measurable` makes that step a
9measurable function of the expression and the value.
10-/
11
12namespace Determinize.Proof.StepTraces
13
14open MeasureTheory ProbabilityTheory Determinize.Spec.Paper Determinize.Proof.StepTraces
15open Determinize.Proof.Paper
16open scoped ProbabilityTheory
17
18noncomputable section
19
20/-- Supply a retained draw directly; other reduction actions have no sampling continuation. -/
21def sampleContinuation (expression : Expr) (value : ℝ) : Expr :=
22 match reduce expression with
23 | .sample _ _ continuation => continuation value
24 | _ => .unit
25
26theorem actionContinuation_measurable {α : Type*} [MeasurableSpace α]
27 {action : α → Action} (family : MeasurableActionFamily α action) :
28 Measurable (fun pair : α × ℝ => match action pair.1 with
29 | .sample _ _ continuation => continuation pair.2
30 | _ => .unit) := by
31 induction family with
32 | next => exact measurable_const
33 | sample draw measurable => exact measurable
34 | stuck => exact measurable_const
35 | @piecewise region _ measurableRegion whenTrue whenFalse trueFamily falseFamily ihTrue ihFalse =>
36 classical
37 convert ihTrue.piecewise (measurableRegion.preimage measurable_fst) ihFalse using 1
38 funext pair
39 by_cases h : pair.1 ∈ region <;> simp [Set.piecewise, h]
40 all_goals infer_instance
41
42theorem sampleContinuation_measurable :
43 Measurable (fun pair : Expr × ℝ => sampleContinuation pair.1 pair.2) := by
44 have localMeasurable (skeleton : Skeleton) :
45 Measurable (fun pair : SkeletonFiber skeleton × ℝ =>
46 sampleContinuation pair.1.val pair.2) :=
47 actionContinuation_measurable
48 (MeasurableActionFamily.reduceFamily primitiveLaws (MeasurableFamily.skeletonFiber skeleton))
49 let selected : Expr × ℝ → Expr := fun pair =>
50 sampleContinuation (MeasurableActionFamily.toSkeletonFiber pair.1.skeleton pair.1).val pair.2
51 have auxiliary : Measurable (fun pair : Skeleton × (Expr × ℝ) =>
52 sampleContinuation (MeasurableActionFamily.toSkeletonFiber pair.1 pair.2.1).val pair.2.2) := by
53 apply measurable_from_prod_countable_right
54 intro skeleton
55 exact (localMeasurable skeleton).comp
56 (((MeasurableActionFamily.measurable_toSkeletonFiber skeleton).comp measurable_fst).prodMk
57 measurable_snd)
58 have selectedMeasurable : Measurable selected :=
59 auxiliary.comp ((measurable_skeleton.comp measurable_fst).prodMk measurable_id)
60 convert selectedMeasurable using 1
61 funext pair
62 change sampleContinuation pair.1 pair.2 =
63 sampleContinuation (MeasurableActionFamily.toSkeletonFiber pair.1.skeleton pair.1).val pair.2
64 rw [MeasurableActionFamily.toSkeletonFiber_coe_of_mem _ _ rfl]
65
66end
