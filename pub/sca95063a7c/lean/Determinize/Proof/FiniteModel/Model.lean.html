5/-- Expected reward returned within `steps` transitions; unfinished paths contribute zero. -/
6def Model.rewardWithin (model : Model) : Nat → Fin model.size → Rat
7 | 0, state => match model.kind state with
8 | .returned reward => reward
9 | _ => 0
10 | steps + 1, state => match model.kind state with
11 | .returned reward => reward
12 | .rejected => 0
13 | .transient => ∑ next, model.transition state next * model.rewardWithin steps next
14
15end Determinize.Spec.FiniteModel

20/-- Increasing the exploration horizon only adds returned output mass. -/
21theorem outputWithin_mono (model : Model) (state : Fin model.size) :
22 Monotone (fun steps => model.outputWithin steps state) := by
23 apply monotone_nat_of_le_succ
24 intro steps
25 induction steps generalizing state with
26 | zero =>
27 cases h : model.kind state <;> simp [Model.outputWithin, h]
28 exact bot_le
29 | succ steps ih =>
30 cases h : model.kind state with
31 | returned reward => simp [Model.outputWithin, h]
32 | rejected => simp [Model.outputWithin, h]
33 | transient =>
34 simp only [Model.outputWithin, h]
35 apply Finset.sum_le_sum
36 intro next _
37 exact smul_le_smul_left _ (ih next)

39theorem returned_outputWithin (model : Model) (state : Fin model.size) (reward : Rat)
40 (returned : model.kind state = .returned reward) (steps : Nat) :
41 model.outputWithin steps state = Measure.dirac (reward : ℝ) := by
42 cases steps <;> simp [Model.outputWithin, returned]

44theorem rejected_outputWithin (model : Model) (state : Fin model.size)
45 (rejected : model.kind state = .rejected) (steps : Nat) :
46 model.outputWithin steps state = 0 := by

58/-- Every finite-horizon output law has finite support. -/
59theorem outputWithin_integrable (model : Model) (steps : Nat) (state : Fin model.size)
60 (f : ℝ → ℝ) : Integrable f (model.outputWithin steps state) := by
61 induction steps generalizing state with
62 | zero =>
63 cases h : model.kind state <;> simp [Model.outputWithin, h]
64 exact integrable_dirac (by simp)
65 | succ steps ih =>
66 cases h : model.kind state with
67 | returned reward =>
68 simpa [Model.outputWithin, h] using
69 (integrable_dirac (by simp) : Integrable f (Measure.dirac (reward : ℝ)))
70 | rejected => simp [Model.outputWithin, h]
71 | transient =>
72 simp only [Model.outputWithin, h]
73 apply integrable_finsetSum_measure.mpr
74 intro next _
75 exact (ih next).smul_measure (by simp)

77/-- Exact rational reward iteration agrees with the real-measure interpretation. -/
78theorem rewardWithin_integral (model : Model) (steps : Nat) (state : Fin model.size) :
79 (∫ value : ℝ, value ∂model.outputWithin steps state) =
80 (model.rewardWithin steps state : ℝ) := by
81 induction steps generalizing state with
82 | zero =>
83 cases h : model.kind state <;> simp [Model.outputWithin, Model.rewardWithin, h]
84 | succ steps ih =>
85 cases h : model.kind state with
86 | returned reward => simp [Model.outputWithin, Model.rewardWithin, h]
87 | rejected => simp [Model.outputWithin, Model.rewardWithin, h]
88 | transient =>
89 simp only [Model.outputWithin, Model.rewardWithin, h]
90 rw [integral_finsetSum_measure (s := Finset.univ)
91 (μ := fun next => ENNReal.ofReal (model.transition state next : ℝ) •
92 model.outputWithin steps next)
93 (fun next _ => (outputWithin_integrable model steps next (fun x => x)).smul_measure
94 (by simp))]
95 simp only [integral_smul_measure, ih, Rat.cast_sum, Rat.cast_mul, smul_eq_mul]
96 apply Finset.sum_congr rfl
97 intro next _
98 rw [ENNReal.toReal_ofReal (by exact_mod_cast model.nonnegative state next)]
