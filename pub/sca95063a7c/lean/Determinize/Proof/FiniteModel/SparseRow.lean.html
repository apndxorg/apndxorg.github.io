6def sparseEdges {n : Nat} (weight : Fin n → Rat) : Array Edge :=
7 (((List.finRange n).filter fun j => decide (0 < weight j)).map fun j => ⟨j.val, weight j⟩).toArray
8
9theorem sparseEdges_weight {n : Nat} (weight : Fin n → Rat) (nonnegative : ∀ j, 0 ≤ weight j)
10 (j : Fin n) :
11 ((sparseEdges weight).toList.map fun edge => if edge.target = j.val then edge.probability else 0).sum =
12 weight j := by
13 simp only [sparseEdges, List.toList_toArray, List.map_map]
14 rw [← List.sum_toFinset _ ((List.nodup_finRange n).filter _), List.toFinset_filter]
15 simp only [List.toFinset_finRange, decide_eq_true_eq]
16 change (∑ k ∈ Finset.univ.filter (fun k => 0 < weight k),
17 if k.val = j.val then weight k else 0) = weight j
18 simp only [← Fin.ext_iff, Finset.sum_ite_eq', Finset.mem_filter, Finset.mem_univ, true_and]
19 split
20 · rfl
21 · rename_i h
22 exact (le_antisymm (le_of_not_gt h) (nonnegative j)).symm
23
24theorem sparseEdges_valid {n : Nat} (weight : Fin n → Rat) :
25 ((sparseEdges weight).toList.map Edge.target).Nodup ∧
26 ∀ edge ∈ (sparseEdges weight).toList, edge.target < n ∧ 0 < edge.probability := by
27 constructor
28 · simp only [sparseEdges, List.toList_toArray, List.map_map]
29 exact ((List.nodup_finRange n).filter _).map Fin.val_injective
30 · intro edge member
31 simp only [sparseEdges, List.toList_toArray, List.mem_map, List.mem_filter,
32 List.mem_finRange, true_and, decide_eq_true_eq] at member
33 obtain ⟨j, positive, rfl⟩ := member
34 exact ⟨j.isLt, positive⟩

36theorem sum_list_comm {n : Nat} (outcomes : List (Rat × State)) (f : Fin n → Rat × State → Rat) :
37 (∑ j, (outcomes.map (f j)).sum) = (outcomes.map fun out => ∑ j, f j out).sum := by
38 induction outcomes with
39 | nil => simp
40 | cons out rest ih => simp [Finset.sum_add_distrib, ih]
