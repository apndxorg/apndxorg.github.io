13def Candidate.state (candidate : Candidate) (i : Fin candidate.states.size) : State :=
14 candidate.states[i]

16def Candidate.row (candidate : Candidate) (i : Fin candidate.states.size) : Row :=
17 candidate.rows[i.val]?.getD ⟨.rejected, #[]⟩
18
19def Candidate.weight (candidate : Candidate) (i j : Fin candidate.states.size) : Rat :=
20 ((candidate.row i).edges.toList.map fun edge =>
21 if edge.target = j.val then edge.probability else 0).sum
22
23/-- Local replay checks the entire probability law, including successors with
24positive mass that are absent from the supplied graph. -/
25def Candidate.RowReplays (candidate : Candidate) (i : Fin candidate.states.size) : Prop :=
26 match step (candidate.state i) with
27 | .error _ => False
28 | .ok (.returned reward) =>
29 (candidate.row i).kind = .returned reward
30 | .ok .rejected =>
31 (candidate.row i).kind = .rejected
32 | .ok (.next _ successors) =>
33 (candidate.row i).kind = .transient ∧
34 (∀ outcome ∈ successors, 0 ≤ outcome.1) ∧
35 (∀ outcome ∈ successors, 0 < outcome.1 →
36 ∃ j : Fin candidate.states.size, candidate.state j = outcome.2) ∧
37 (∀ j : Fin candidate.states.size, candidate.weight i j =
38 (successors.map fun outcome =>
39 if outcome.2 = candidate.state j then outcome.1 else 0).sum)

50structure Candidate.MatrixValid (candidate : Candidate) : Prop where
51 nonnegative : ∀ i j, 0 ≤ candidate.weight i j
52 normalized : ∀ i, ∑ j, candidate.weight i j = 1
53 absorbing : ∀ i, (candidate.row i).kind ≠ .transient →
54 ∀ j, candidate.weight i j = if i = j then 1 else 0
55
56instance (candidate : Candidate) : Decidable candidate.MatrixValid :=
57 decidable_of_iff
58 ((∀ i j, 0 ≤ candidate.weight i j) ∧
59 (∀ i, ∑ j, candidate.weight i j = 1) ∧
60 (∀ i, (candidate.row i).kind ≠ .transient →
61 ∀ j, candidate.weight i j = if i = j then 1 else 0))
62 ⟨fun ⟨a, b, c⟩ => ⟨a, b, c⟩, fun ⟨a, b, c⟩ => ⟨a, b, c⟩⟩
63
64/-- Sparse representation checks additionally reject out-of-range, duplicate,
65and nonpositive edges, even when they cancel in the induced matrix. -/
66def Candidate.EdgesValid (candidate : Candidate) (i : Fin candidate.states.size) : Prop :=
67 ((candidate.row i).edges.toList.map Edge.target).Nodup ∧
68 ∀ edge ∈ (candidate.row i).edges.toList,
69 edge.target < candidate.states.size ∧ 0 < edge.probability

71instance (candidate : Candidate) (i : Fin candidate.states.size) :
72 Decidable (candidate.EdgesValid i) := inferInstanceAs (Decidable (_ ∧ _))

89structure Candidate.ReplayValid (candidate : Candidate) (source : Core) (subject : Subject) : Prop where
90 rows_size : candidate.rows.size = candidate.states.size
91 initial_lt : candidate.initial < candidate.states.size
92 aligned : candidate.Aligned source subject
93 matrix : candidate.MatrixValid
94 edges : ∀ i, candidate.EdgesValid i
95 replays : ∀ i, candidate.RowReplays i

97instance (candidate : Candidate) (source : Core) (subject : Subject) :
98 Decidable (candidate.ReplayValid source subject) :=
99 decidable_of_iff
100 (candidate.rows.size = candidate.states.size ∧ candidate.initial < candidate.states.size ∧
101 candidate.Aligned source subject ∧ candidate.MatrixValid ∧
102 (∀ i, candidate.EdgesValid i) ∧ (∀ i, candidate.RowReplays i))
103 ⟨fun ⟨a, b, c, d, e, f⟩ => ⟨a, b, c, d, e, f⟩,
104 fun ⟨a, b, c, d, e, f⟩ => ⟨a, b, c, d, e, f⟩⟩

106/-- This constructs a finite rational model, but supplies no `Matches` proof. -/
107abbrev Candidate.toModel (candidate : Candidate) {source : Core} {subject : Subject}
108 (valid : candidate.ReplayValid source subject) : Model where
109 size := candidate.states.size
110 initial := ⟨candidate.initial, valid.initial_lt⟩
111 kind := fun i => (candidate.row i).kind
112 transition := candidate.weight
113 nonnegative := valid.matrix.nonnegative
114 normalized := valid.matrix.normalized

116theorem replay_successor_covered (candidate : Candidate) {source : Core} {subject : Subject}
117 (valid : candidate.ReplayValid source subject) (i : Fin candidate.states.size)
118 (evidence : Evidence) (successors : List (Rat × State))
119 (action : step (candidate.state i) = .ok (.next evidence successors))
120 (outcome : Rat × State) (member : outcome ∈ successors) (positive : 0 < outcome.1) :
121 ∃ j : Fin candidate.states.size, candidate.state j = outcome.2 := by
122 have localValid := valid.replays i
123 simp only [Candidate.RowReplays, action] at localValid
124 exact localValid.2.2.1 outcome member positive

126theorem replay_transition_weight (candidate : Candidate) {source : Core} {subject : Subject}
127 (valid : candidate.ReplayValid source subject) (i j : Fin candidate.states.size)
128 (evidence : Evidence) (successors : List (Rat × State))
129 (action : step (candidate.state i) = .ok (.next evidence successors)) :
130 (candidate.toModel valid).transition i j =
131 (successors.map fun outcome =>
132 if outcome.2 = candidate.state j then outcome.1 else 0).sum := by
133 have localValid := valid.replays i
134 simp only [Candidate.RowReplays, action] at localValid
135 exact localValid.2.2.2 j

137/-- Reachability along positive-probability transitions of the executable machine. -/
138inductive MachineReachable (initial : State) : State → Prop where
139 | initial : MachineReachable initial initial
140 | next {before after : State} {probability : Rat} {evidence : Evidence}
141 {successors : List (Rat × State)} :
142 MachineReachable initial before →
143 step before = .ok (.next evidence successors) →
144 (probability, after) ∈ successors → 0 < probability →
145 MachineReachable initial after

147theorem replay_initial (candidate : Candidate) {source : Core} {subject : Subject}
148 (valid : candidate.ReplayValid source subject) :
149 candidate.state (candidate.toModel valid).initial = initialState source subject := by
150 have aligned := valid.aligned.initial
151 simpa [valid.initial_lt, Candidate.state, Candidate.toModel] using aligned

153/-- Coverage holds for paths of arbitrary length, not just the explorer's horizon. -/
154theorem replay_reachable_covered (candidate : Candidate) {source : Core} {subject : Subject}
155 (valid : candidate.ReplayValid source subject) {state : State}
156 (reachable : MachineReachable (initialState source subject) state) :
157 ∃ i : Fin candidate.states.size, candidate.state i = state := by
158 induction reachable with
159 | initial => exact ⟨(candidate.toModel valid).initial, replay_initial candidate valid⟩
160 | next previous action member positive ih =>
161 obtain ⟨i, hi⟩ := ih
162 exact replay_successor_covered candidate valid i _ _ (hi ▸ action) _ member positive

164theorem replay_no_failure (candidate : Candidate) {source : Core} {subject : Subject}
165 (valid : candidate.ReplayValid source subject) (i : Fin candidate.states.size)
166 (failure : Failure) : step (candidate.state i) ≠ .error failure := by
167 intro action
168 have localValid := valid.replays i
169 simp [Candidate.RowReplays, action] at localValid

171theorem replay_reachable_no_failure (candidate : Candidate) {source : Core} {subject : Subject}
172 (valid : candidate.ReplayValid source subject) {state : State}
173 (reachable : MachineReachable (initialState source subject) state) (failure : Failure) :
174 step state ≠ .error failure := by
175 obtain ⟨i, hi⟩ := replay_reachable_covered candidate valid reachable
176 rw [← hi]
177 exact replay_no_failure candidate valid i failure
