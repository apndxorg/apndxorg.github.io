4/-!
5# The expectation theorem
6
7`Spec.mainThm` (exported as `Theorems.expectationPreservation`) follows from trace
8soundness. Under a `TraceFactorization`, the source output law is the mixture of the fibers
9over the trace law (`TraceFactorization.source_law`) and the target output law is the
10pushforward of the trace law along the fiber means (`TraceFactorization.target_law`), so
11integrating the fibers preserves a finite expectation (`MeanOnTraces.finite_expectation`);
12`finiteExpectationSoundness` instantiates this with the factorization that `meanOnTraces`
13provides for a typed source that does not get stuck.
14-/
15
16namespace Determinize.Proof.Traces
17open MeasureTheory ProbabilityTheory Determinize.Spec.Paper Determinize.Spec.Traces
18open scoped ProbabilityTheory
19
20/-- The source output law under a trace factorization: the mixture of the fibers over the
21trace law. -/
22theorem TraceFactorization.source_law {source target : Expr} {fiber : Kernel Trace ℝ}
23 {output : Trace → ℝ} (factor : TraceFactorization source target fiber output) :
24 bigStepMeasure source = fiber ∘ₘ traceLaw source := by
25 obtain ⟨markov, -, sourceJoint, -, -⟩ := factor
26 have := markov
27 rw [← Proof.Traces.correspondence source, sourceJoint]
28 exact Measure.snd_compProd (traceLaw source) fiber
29
30/-- The target output law under a trace factorization: the pushforward of the trace law along
31`output`. -/
32theorem TraceFactorization.target_law {source target : Expr} {fiber : Kernel Trace ℝ}
33 {output : Trace → ℝ} (factor : TraceFactorization source target fiber output) :
34 bigStepMeasure target = (traceLaw source).map output := by
35 obtain ⟨-, measurableOutput, -, targetJoint, -⟩ := factor
36 rw [← Proof.Traces.correspondence target, targetJoint,
37 Measure.map_map measurable_snd (show Measurable (fun trace => (trace, output trace)) from
38 measurable_id.prodMk measurableOutput)]
39 rfl
40
41/-- Integrating the trace fibers preserves finite expectation. -/
42theorem MeanOnTraces.finite_expectation {source target : Expr} (sound : MeanOnTraces source target)
43 (integrable : Integrable id (bigStepMeasure source)) :
44 Integrable id (bigStepMeasure target) ∧
45 (∫ value : ℝ, value ∂bigStepMeasure source) = ∫ value : ℝ, value ∂bigStepMeasure target := by
46 obtain ⟨fiber, factor⟩ := sound
47 let output := kernelMean fiber
48 have sourceEq := factor.source_law
49 have targetEq := factor.target_law
50 obtain ⟨markov, measurableOutput, -, -, valid⟩ := factor
51 have := markov
52 rw [sourceEq] at integrable ⊢
53 rw [targetEq]
54 have normIntegrable := Measure.integrable_integral_norm_of_integrable_comp integrable
55 have outputIntegrable : Integrable output (traceLaw source) := by
56 apply normIntegrable.mono' measurableOutput.aestronglyMeasurable
57 filter_upwards [valid] with trace good
58 rw [good.2]
59 exact norm_integral_le_integral_norm _
60 refine ⟨(integrable_map_measure aestronglyMeasurable_id measurableOutput.aemeasurable).2
61 outputIntegrable, ?_⟩
62 change (∫ value : ℝ, id value ∂fiber ∘ₘ traceLaw source) =
63 ∫ value : ℝ, id value ∂(traceLaw source).map output
64 rw [integral_map measurableOutput.aemeasurable aestronglyMeasurable_id]
65 rw [Measure.comp_eq_comp_const_apply] at integrable ⊢
66 rw [Kernel.integral_comp integrable]
67 simp only [Kernel.const_apply]
68 exact integral_congr_ae (valid.mono fun _ good => good.2.symm)
69
70end Determinize.Proof.Traces

74/-- The public expectation theorem follows from operational trace soundness. -/
75theorem finiteExpectationSoundness : Determinize.Spec.mainThm := by
76 intro program typed sourceSafe sourceIntegrable
77 rcases Determinize.Proof.Traces.meanOnTraces .E program typed
78 sourceSafe with ⟨targetSafe, traces⟩
79 rcases traces.finite_expectation sourceIntegrable with ⟨targetIntegrable, expectation⟩
80 exact ⟨targetSafe,
81 targetIntegrable, expectation⟩
