4/-!
5# Expression coordinates for measurability proofs
6
7An expression is encoded as a real-free skeleton together with the list of its
8real literals. The skeleton carries the discrete sigma-algebra; the real
9coordinates are measured through their length and their zero-padded entries,
10exactly like the trace layer measures a trace. This structure is absent from
11the direct evaluators and the public theorem statements.
12-/
13
14namespace Determinize.Spec.Paper
15
16/-- The same syntax with real values erased. -/
17abbrev Skeleton := Expr Unit
18
19namespace Expr
20
21def skeleton : Expr → Skeleton
22 | .discrete kind d => .discrete kind d.skeleton
23 | .bvar i => .bvar i | .reject => .reject | .unit => .unit | .bool b => .bool b
24 | .real _ => .real () | .lam b => .lam b.skeleton
25 | .fix b => .fix b.skeleton | .app f x => .app f.skeleton x.skeleton
26 | .pair l r => .pair l.skeleton r.skeleton | .fst x => .fst x.skeleton
27 | .snd x => .snd x.skeleton | .inl x => .inl x.skeleton
28 | .inr x => .inr x.skeleton
29 | .matchSum x l r => .matchSum x.skeleton l.skeleton r.skeleton
30 | .nil => .nil | .cons h t => .cons h.skeleton t.skeleton
31 | .matchList x n c => .matchList x.skeleton n.skeleton c.skeleton
32 | .ite c t e => .ite c.skeleton t.skeleton e.skeleton
33 | .letE x b => .letE x.skeleton b.skeleton
34 | .neg x => .neg x.skeleton
35 | .add l r => .add l.skeleton r.skeleton | .mul l r => .mul l.skeleton r.skeleton
36 | .div l r => .div l.skeleton r.skeleton | .lt l r => .lt l.skeleton r.skeleton
37 | .uniform k l r => .uniform k l.skeleton r.skeleton
38 | .gaussian k l r => .gaussian k l.skeleton r.skeleton
39 | .poisson k x => .poisson k x.skeleton
40 | .bernoulli k x => .bernoulli k x.skeleton
41 | .exponential k x => .exponential k x.skeleton
42 | .beta k l r => .beta k l.skeleton r.skeleton
43 | .gamma k l r => .gamma k l.skeleton r.skeleton
44
45def realCoordinates : Expr → List ℝ
46 | .real value => [value]
47 | .lam x | .fix x | .fst x | .snd x | .inl x
48 | .inr x | .neg x => x.realCoordinates
49 | .app l r | .pair l r | .cons l r | .add l r | .mul l r
50 | .div l r | .lt l r => l.realCoordinates ++ r.realCoordinates
51 | .matchSum x l r | .ite x l r =>
52 x.realCoordinates ++ l.realCoordinates ++ r.realCoordinates
53 | .matchList x n c => x.realCoordinates ++ n.realCoordinates ++ c.realCoordinates
54 | .letE x b => x.realCoordinates ++ b.realCoordinates
55 | .uniform _ l r | .gaussian _ l r | .beta _ l r | .gamma _ l r =>
56 l.realCoordinates ++ r.realCoordinates
57 | .poisson _ x | .bernoulli _ x | .exponential _ x | .discrete _ x => x.realCoordinates
58 | _ => []
59
60/-- The number of literals in a list value of literals, a cons-chain of literals ending in
61`nil`, read off the syntax alone; `none` for any other expression. On an `Expr` this is the
62length of the list `realListValue?` returns, and it only depends on the skeleton, so the arity
63of a `discrete` site is a function of the skeleton of its evaluated weights. -/
64def literalListArity? {Literal : Type} : Expr Literal → Option Nat
65 | .nil => some 0
66 | .cons (.real _) tail => (literalListArity? tail).map (· + 1)
67 | _ => none
68
69theorem literalListArity?_skeleton (expression : Expr) :
70 expression.skeleton.literalListArity? = expression.literalListArity? := by
71 induction expression with
72 | cons head tail _ ih => cases head <;> simp [literalListArity?, skeleton, ih]
73 | _ => simp [literalListArity?, skeleton]
74
75end Expr
76
77/-- `realListValue?` succeeds exactly on a cons-chain of real literals, with one real per
78literal (`Expr.literalListArity?`). -/
79theorem realListValue?_map_length (expression : Expr) :
80 (realListValue? expression).map List.length = expression.literalListArity? := by
81 induction expression with
82 | cons head tail _ ih =>
83 cases head <;> simp only [realListValue?, realValue?, Expr.literalListArity?, ← ih] <;>
84 cases realListValue? tail <;> simp
85 | _ => simp [realListValue?, Expr.literalListArity?]
86
87/-- The reals of a list value of literals are its real coordinates, in order. -/
88theorem realCoordinates_of_realListValue? {expression : Expr} {values : List ℝ}
89 (equation : realListValue? expression = some values) :
90 expression.realCoordinates = values := by
91 induction expression generalizing values with
92 | cons head tail _ ih =>
93 cases head <;> simp only [realListValue?, realValue?] at equation <;>
94 cases tailEq : realListValue? tail <;> simp [tailEq] at equation
95 rw [← equation, Expr.realCoordinates, Expr.realCoordinates, ih tailEq]
96 rfl
97 | _ => simp_all [realListValue?, Expr.realCoordinates]
98
99theorem realValue?_determinize (expression : Expr) :
100 realValue? expression.determinize = realValue? expression := by
101 cases expression <;> simp [Expr.determinize, realValue?]
102
103/-- Determinization does not change the weights a `discrete` site reads: it maps literals,
104`cons` and `nil` to themselves. -/
105theorem realListValue?_determinize (expression : Expr) :
106 realListValue? expression.determinize = realListValue? expression := by
107 induction expression with
108 | cons head tail _ ih => simp [Expr.determinize, realListValue?, realValue?_determinize, ih]
109 | _ => simp [Expr.determinize, realListValue?]
110
111structure RealCoordinates where values : List ℝ
112
113/-- Length and zero-padded coordinates, mirroring the measurable structure of traces
114in `Determinize.Spec.Traces`. The length is recorded because padding alone is not injective. -/
115def RealCoordinates.code (coordinates : RealCoordinates) : Nat × (Nat → ℝ) :=
116 (coordinates.values.length, fun index => coordinates.values.getD index 0)
117
118instance : MeasurableSpace RealCoordinates :=
119 MeasurableSpace.comap RealCoordinates.code inferInstance
120
121/-- A discrete skeleton paired with real coordinates. Every expression with a fixed
122skeleton has a fixed number of real literals, so under `code` its coordinates are that
123length together with zero-padded entries, measured as in the trace layer. -/
124abbrev Code := Skeleton × RealCoordinates
125
126def code (expression : Expr) : Code :=
127 (expression.skeleton, ⟨expression.realCoordinates⟩)
128
129instance : MeasurableSpace Skeleton := ⊤
130instance : MeasurableSpace Expr := MeasurableSpace.comap code inferInstance
131
132def terminalFloatSet : Set Expr := {expression | ∃ value, expression = .real value}
133
134def terminalFloatValue : Expr → ℝ
135 | .real value => value
136 | _ => 0
