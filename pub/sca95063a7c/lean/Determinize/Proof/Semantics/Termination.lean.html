11private theorem returned_add_running_eq_step_mass (depth : Nat) (typed : Typed [] expression (.float affinity)) :
12 Cumulative.outputMeasure depth expression Set.univ + runningProbabilityAt depth expression =
13 nStepMeasure paperStepKernel depth expression Set.univ := by
14 induction depth generalizing expression with
15 | zero =>
16 by_cases value : expression.isValue = true
17 · obtain ⟨r, rfl⟩ := Typing.typed_real_value typed value
18 simp [Cumulative.outputMeasure, runningProbabilityAt, nStepMeasure, Expr.isValue]
19 · cases expression <;> simp_all [Cumulative.outputMeasure, runningProbabilityAt, nStepMeasure, Expr.isValue]
20 | succ depth ih =>
21 by_cases value : expression.isValue = true
22 · rw [nStepMeasure_univ_eq_one_of_value paperStepKernel _ _ value]
23 obtain ⟨r, rfl⟩ := Typing.typed_real_value typed value
24 have output : ∀ n, Cumulative.outputMeasure n (.real r) = Measure.dirac r := by
25 intro n
26 induction n with
27 | zero => rfl
28 | succ n ih => simpa [Cumulative.outputMeasure, reduce] using ih
29 simp [output, runningProbabilityAt, Expr.isValue]
30 · simp only [runningProbabilityAt, value, Bool.false_eq_true, ↓reduceIte]
31 have actionTyped := Typing.reduce_typed_closed typed
32 cases reduction : reduce expression with
33 | stuck =>
34 rw [nStepMeasure_succ_eq_firstStep, paperStepKernel.kernel_eq_stepMeasure]
35 simp [stepMeasure, reduction, Action.measure, Cumulative.outputMeasure]
36 | next next =>
37 rw [reduction] at actionTyped
38 cases actionTyped with
39 | next nextTyped =>
40 rw [nStepMeasure_succ_next_univ paperStepKernel _ _ _ reduction]
41 simpa [Cumulative.outputMeasure, reduction] using ih nextTyped
42 | sample site fiber continuation =>
43 rw [reduction] at actionTyped
44 cases actionTyped with
45 | sample continuationTyped =>
46 have hm : Measurable (fun r => Cumulative.outputMeasure depth (continuation r) Set.univ) :=
47 (Measure.measurable_coe MeasurableSet.univ).comp
48 (measurable_sample_cumulative depth expression site fiber continuation reduction)
49 change Cumulative.outputMeasure (depth + 1) expression Set.univ +
50 (∫⁻ r, runningProbabilityAt depth (continuation r) ∂fiber) = _
51 rw [Cumulative.outputMeasure, reduction,
52 Measure.bind_apply MeasurableSet.univ
53 (measurable_sample_cumulative depth expression site fiber continuation reduction).aemeasurable,
54 ← lintegral_add_left hm,
55 nStepMeasure_succ_sample_univ paperStepKernel _ _ _ _ reduction]
56 exact lintegral_congr fun r => ih (continuationTyped r)

58private theorem total_antitone (expression : Expr) :
59 Antitone (fun n => nStepMeasure paperStepKernel n expression Set.univ) := by
60 apply antitone_nat_of_succ_le
61 intro n
62 rw [nStepMeasure, Measure.bind_apply MeasurableSet.univ paperStepKernel.kernel.aemeasurable]
63 calc
64 (∫⁻ current, paperStepKernel.kernel current Set.univ ∂nStepMeasure paperStepKernel n expression) ≤
65 ∫⁻ _, 1 ∂nStepMeasure paperStepKernel n expression :=
66 lintegral_mono fun current => paperStepKernel.mass_le_one current
67 _ = _ := lintegral_one

98theorem returnOrDiverge : Determinize.Spec.returnOrDivergeThm := by
99 intro affinity expression typed safe
100 have bounded : Determinize.Spec.Paper.bigStepMeasure expression Set.univ ≤ 1 := by
101 rw [output_mass_eq_iSup_cumulative]

114private theorem return_diverge_le_total (typed : Typed [] expression (.float affinity)) (n : Nat) :
115 Determinize.Spec.Paper.bigStepMeasure expression Set.univ + divergenceProbability expression ≤
116 nStepMeasure paperStepKernel n expression Set.univ := by
117 rw [output_mass_eq_iSup_cumulative, ENNReal.iSup_add]
118 apply iSup_le
119 intro k
120 calc
121 Cumulative.outputMeasure k expression Set.univ + divergenceProbability expression ≤
122 Cumulative.outputMeasure (max k n) expression Set.univ + runningProbabilityAt (max k n) expression :=
123 add_le_add ((direct_cumulative_mono expression (Nat.le_max_left k n)) Set.univ) (iInf_le _ _)
124 _ = _ := returned_add_running_eq_step_mass _ typed
125 _ ≤ _ := total_antitone expression (Nat.le_max_right k n)

128 DomainSafe expression ↔
