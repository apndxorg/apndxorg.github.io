7/-- A rejection never contributes a returned float at any execution depth. -/
8theorem output_at_reject (fuel : Nat) : outputMeasureAt fuel .reject = 0 := by
9 induction fuel with
10 | zero => rfl
11 | succ fuel ih => simpa [outputMeasureAt, reduce, Expr.isValue] using ih
12
13theorem reject_zero : bigStepMeasure .reject = 0 := by
14 simp [bigStepMeasure, output_at_reject]
15
16/-- Rejection is intentional loss of output mass, not an invalid primitive call. -/
17theorem reject_safe : DomainSafe .reject := by
18 intro fuel
19 induction fuel with
20 | zero => trivial
21 | succ fuel ih => simpa [DomainSafeAt, Expr.isValue, reduce] using ih
22
23theorem output_at_let_reject (fuel : Nat) (body : Expr) :
24 outputMeasureAt fuel (.letE .reject body) = 0 := by
25 induction fuel with
26 | zero => rfl
27 | succ fuel ih => simpa [outputMeasureAt, reduce, Expr.isValue, Action.wrap] using ih
28
29theorem let_reject_zero (body : Expr) : bigStepMeasure (.letE .reject body) = 0 := by
30 simp [bigStepMeasure, output_at_let_reject]
31
32theorem failed_observation_zero : bigStepMeasure (.ite (.bool false) .unit .reject) = 0 := by
33 unfold bigStepMeasure
34 have h (fuel : Nat) : outputMeasureAt fuel (.ite (.bool false) .unit .reject) = 0 := by
35 cases fuel with
36 | zero => rfl
37 | succ fuel => simp [outputMeasureAt, reduce, Expr.isValue, output_at_reject]
38 simp [h]

40theorem determinize_reject : Expr.determinize (.reject : Expr) = .reject := rfl
