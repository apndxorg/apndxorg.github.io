177theorem hasVar_subst (h : HasVar (before ++ binder :: suffix) index ty) :
178 (index = before.length ∧ ty = binder) ∨
179 (index ≠ before.length ∧ HasVar (before ++ suffix)
180 (if before.length < index then index - 1 else index) ty) := by
181 induction before generalizing index with
182 | nil =>
183 cases h with
184 | head => exact .inl ⟨rfl, rfl⟩
185 | tail h =>
186 right
187 exact ⟨by simp only [List.length_nil]; omega, by simpa using h⟩
188 | cons head before ih =>
189 cases h with
190 | head =>
191 right
192 simp only [List.cons_append, List.length_cons]
193 exact ⟨by omega, .head⟩
194 | tail h =>
195 rcases ih h with equal | shifted
196 · left
197 exact ⟨congrArg Nat.succ equal.1, equal.2⟩
198 · right
199 rcases shifted with ⟨notEqual, shifted⟩
200 rename_i index
201 constructor
202 · simp only [List.length_cons]
203 omega
204 simp only [List.cons_append, List.length_cons, Nat.succ_lt_succ_iff]
205 by_cases condition : before.length < index
206 · simp only [condition, ↓reduceIte]

373 | stuck : ActionTyped ty .stuck
374 | next : Typed [] expression ty → ActionTyped ty (.next expression)
375 | sample : (∀ value, Typed [] (continuation value) ty) →
376 ActionTyped ty (.sample site fiber continuation)
