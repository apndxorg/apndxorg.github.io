8private def rational (q : Rat) : String :=
9 if q.den == 1 then toString q.num else s!"{q.num}/{q.den}"
10private def leanRat (q : Rat) : String := s!"(({q.num} : Rat) / {q.den})"
11private def keyTree : Nat → List Nat → Nat → String
12 | 0, _, _ => "0"
13 | _+1, [], _ => "0"
14 | _+1, [value], _ => toString value
15 | fuel+1, values, offset =>
16 let half := values.length / 2
17 s!"(if i.val < {offset+half} then {keyTree fuel (values.take half) offset} else {keyTree fuel (values.drop half) (offset+half)})"

19private def proofTable (stem claim sizeName allName : String) (size : Nat) : String :=
20 let proofs := String.join ((List.range size).map fun i =>
21 s!"\ntheorem {stem}_{i} : {claim} (⟨{i}, by decide +kernel⟩ : Fin {sizeName}) := by\n decide +kernel\n")
22 let entries := String.intercalate ", " ((List.range size).map fun i =>
23 s!"⟨⟨{i}, by decide +kernel⟩, {stem}_{i}⟩")
24 proofs ++
25 s!"\ndef {stem}Evidence : Vector (Subtype (fun i : Fin {sizeName} => {claim} i)) {sizeName} := ⟨#[{entries}], by rfl⟩\n" ++
26 s!"theorem {stem}Indices : ∀ i : Fin {sizeName}, ({stem}Evidence[i]).val = i := by decide +kernel\n" ++
27 s!"theorem {allName} (i : Fin {sizeName}) : {claim} i := ({stem}Indices i) ▸ ({stem}Evidence[i]).property\n"

89def replayCertificateText (source : Checking.Core) (subject : Subject) (candidate : Candidate) : String :=
90 let keys := keyTree candidate.states.size (candidate.states.toList.map stateFingerprint) 0
91 let indices := candidate.states.toList.map fun state =>
92 let destinations : List Nat := match step state with
93 | .ok (.next _ outcomes) => outcomes.map fun (outcome : Rat × State) =>
94 (candidate.states.toList.findIdx? fun s => decide (s = outcome.2)).getD 0
95 | _ => []
96 "#[" ++ String.intercalate ", " (destinations.map toString) ++ "]"
97 let indices := "#[" ++ String.intercalate ", " indices ++ "]"
98 let rowProofs := proofTable "row" "rowClaim" "candidate.states.size" "allRows" candidate.states.size
99 (candidateText candidate).replace "import Determinize.Finite.Graph"
100 "import Determinize.Checking.FiniteModel" ++
101 s!"\ndef checkedSource : Expr Rat := {Frontend.leanExpression source}\n" ++
102 s!"def checkedSubject : Subject := {reprStr subject}\n" ++
103 s!"\ndef stateKeys (i : Fin candidate.states.size) : Nat := {keys}\n" ++
104 s!"def successorIndices : Vector (Array Nat) candidate.states.size := ⟨{indices}, by rfl⟩\n" ++
105 "\nabbrev rowClaim (i : Fin candidate.states.size) : Prop := candidate.IndexedStateValid\n" ++
106 " (fun i => stateKeys i) (fun i k => successorIndices[i][k]!) i\n" ++ rowProofs ++
107 "\ntheorem indexedReplay : candidate.IndexedReplayValid checkedSource checkedSubject\n" ++

109 " indexedStates_valid candidate checkedSource checkedSubject _ _ (by decide +kernel) allRows\n" ++
110 "\ntheorem machineReplay : candidate.ReplayValid checkedSource checkedSubject :=\n" ++
111 " indexedReplay_valid candidate checkedSource checkedSubject _ _ indexedReplay\n\nabbrev model : Model := candidate.toModel machineReplay\n" ++
112 "\ntheorem modelMatches : model.Matches (checkedSubject.program checkedSource) :=\n" ++
113 " Determinize.Proof.FiniteModel.replay_matches candidate machineReplay\n" ++

124/-- Structural validation only; this does not check machine transition correctness. -/
125def render (candidate : Candidate) : Except String Files := do
126 let size := candidate.states.size
127 if size == 0 || candidate.initial ≥ size || candidate.rows.size != size then
128 throw "invalid candidate dimensions"
129 let mut transitions := "dtmc\n"
130 let mut labels := "#DECLARATION\ninit returned rejected done\n#END\n"
131 let mut positiveRewards := ""
132 let mut negativeRewards := ""
133 for i in [:size] do
134 let row := candidate.rows[i]?.getD ⟨.rejected, #[]⟩
135 let mut mass := (0 : Rat)
136 let mut targets : List Nat := []
137 for edge in row.edges do
138 if edge.target ≥ size || edge.probability ≤ 0 || targets.contains edge.target then
139 throw "invalid or duplicate candidate edge"
140 targets := edge.target :: targets
141 mass := mass + edge.probability
142 if mass != 1 then throw "candidate row is not normalized"
143 let mut names := if i == candidate.initial then ["init"] else []
144 match row.kind with
145 | .transient =>
146 for edge in row.edges do
147 transitions := transitions ++ s!"{i} {edge.target} {rational edge.probability}\n"
148 | terminal =>
149 unless row.edges == #[⟨i,1⟩] do throw "candidate terminal state is not absorbing"
150 transitions := transitions ++ s!"{i} {size} 1\n"
151 match terminal with
152 | .returned reward =>
153 names := names ++ ["returned"]
154 if reward > 0 then positiveRewards := positiveRewards ++ s!"{i} {rational reward}\n"
155 if reward < 0 then negativeRewards := negativeRewards ++ s!"{i} {rational (-reward)}\n"
156 | _ => names := names ++ ["rejected"]
157 if !names.isEmpty then labels := labels ++ s!"{i} {String.intercalate " " names}\n"
158 transitions := transitions ++ s!"{size} {size} 1\n"
159 labels := labels ++ s!"{size} done\n"
160 return ⟨candidateText candidate, transitions, labels, positiveRewards, negativeRewards⟩

171 IO.FS.writeFile (outputPath.toString ++ suffix) content

219def writeResult (outputPath : System.FilePath) (source : Checking.Core) (subject : Subject)
220 (candidate : Candidate) (valid : candidate.ReplayValid source subject)
221 (limits : SolveLimits := {}) : IO Rat := do
222 let model := candidate.toModel valid
223 let certified ← IO.ofExcept (solveStatistics model limits)
224 let termination ← IO.ofExcept (solveTermination model certified limits)
225 let outcomes := termination.val.statistics model
226 let certificate := certified.val
227 let statistics := certificate.statistics model
228 let rankBound := (List.ofFn certificate.rank).foldl max 0
229 let optional := fun value : Option Rat => match value with
230 | none => Lean.Json.null | some q => Lean.toJson (rational q)
231 let metadata := Lean.Json.mkObj [
232 ("answer", Lean.toJson (rational statistics.firstMoment)),
233 ("return_mass", Lean.toJson (rational statistics.returnMass)),
234 ("rejection_probability", Lean.toJson (rational outcomes.rejectionProbability)),
235 ("divergence_probability", Lean.toJson (rational outcomes.divergenceProbability)),
236 ("second_moment", Lean.toJson (rational statistics.secondMoment)),
237 ("conditional_mean", optional statistics.conditionalMean),
238 ("conditional_variance", optional statistics.conditionalVariance),
239 ("subject", Lean.toJson (if subject == .source then "source" else "determinized")),
240 ("states", Lean.toJson model.size),
241 ("rank_bound", Lean.toJson rankBound)]
242 write outputPath source subject candidate valid
243 IO.FS.writeFile (outputPath.toString ++ ".result.lean")
244 (resultCertificateText source subject candidate model termination.val)
245 IO.FS.writeFile (outputPath.toString ++ ".result.json") (metadata.pretty ++ "\n")
246 return statistics.firstMoment
