6/-- Initial exporter's eligibility rule for an evaluated sampling call. This does
7not check arity or the parameter domain. All mean formulas are rational on rational
8parameters; stochastic export initially supports only Bernoulli and finite discrete. -/
9def supportedDraw : DistributionAction → Op → Bool
10 | .mean, _ => true
11 | .sample _, .bernoulli | .sample _, .discrete _ => true
12 | .sample _, _ => false

14end Determinize.Finite
